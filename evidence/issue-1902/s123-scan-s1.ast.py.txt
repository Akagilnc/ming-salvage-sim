#!/usr/bin/env python3
"""S1 FULL: all tracked prod FunctionDef/AsyncFunctionDef/ClassDef/Assign(const)
with zero non-test Load/Attribute consumers. No sample-name filter for emission.
Framework registration / dynamic dispatch marked for human KEEP.
Temporary; not permanent scanner.
"""
from __future__ import annotations
import ast, json, os, re, subprocess, sys
from collections import Counter, defaultdict
from pathlib import Path

ROOT = Path(os.environ.get("SCAN_ROOT", ".")).resolve()
os.chdir(ROOT)
OUTDIR = Path(os.environ.get("SCAN_OUT", "/tmp/ming1902-s123-full"))
OUTDIR.mkdir(parents=True, exist_ok=True)

tracked = subprocess.check_output(["git", "ls-files", "*.py"], text=True).splitlines()

def is_test_path(p: str) -> bool:
    return p.startswith("tests/") or p.endswith("_test.py") or "/tests/" in p or p.startswith("spike")

# Collect string literals that look like dispatch targets
string_mentions = defaultdict(list)
decorator_routes = defaultdict(list)
defs = defaultdict(list)
const_assigns = defaultdict(list)
refs = defaultdict(list)

for path in tracked:
    try:
        src = Path(path).read_text(encoding="utf-8")
        tree = ast.parse(src, filename=path)
    except Exception as e:
        print(f"WARN {path}: {e}", file=sys.stderr)
        continue
    for node in ast.walk(tree):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            defs[node.name].append({
                "file": path, "line": node.lineno,
                "kind": "async" if isinstance(node, ast.AsyncFunctionDef) else "func",
                "decorators": [ast.dump(d)[:120] for d in node.decorator_list],
            })
            for d in node.decorator_list:
                dump = ast.dump(d)
                if any(x in dump for x in ("app.get", "app.post", "app.put", "app.patch", "app.delete",
                                            "router.", "classmethod", "staticmethod", "property")):
                    decorator_routes[node.name].append({"file": path, "line": node.lineno, "dec": dump[:160]})
        elif isinstance(node, ast.ClassDef):
            defs[node.name].append({"file": path, "line": node.lineno, "kind": "class", "decorators": []})
        elif isinstance(node, ast.Assign):
            # module-level ALL_CAPS / leading underscore const
            if hasattr(node, "lineno"):
                for t in node.targets:
                    if isinstance(t, ast.Name) and (t.id.isupper() or t.id.startswith("_")):
                        const_assigns[t.id].append({"file": path, "line": node.lineno, "kind": "const"})
        elif isinstance(node, ast.Name) and isinstance(node.ctx, ast.Load):
            refs[node.id].append({"file": path, "line": node.lineno, "test": is_test_path(path)})
        elif isinstance(node, ast.Attribute) and isinstance(node.ctx, ast.Load):
            refs[node.attr].append({"file": path, "line": node.lineno, "test": is_test_path(path)})
        elif isinstance(node, ast.Constant) and isinstance(node.value, str):
            v = node.value
            if len(v) >= 3 and (v.startswith("_") or v.isidentifier()):
                string_mentions[v].append({"file": path, "line": getattr(node, "lineno", 0), "test": is_test_path(path)})

KEEP_SHARED_SAMPLES = {
    "secret_order_can_land", "secret_order_landing_gaps", "CliChat",
    "_fake_completion", "extract_draft_intent", "extract_draft_intent_payload",
    "resubmit_draft_admission_payload", "heal_draft_admission",
    "update_secret_order_by_id", "create_secret_order",
}

members = []
all_names = set(defs) | set(const_assigns)
for name in sorted(all_names):
    locs = defs.get(name, []) + const_assigns.get(name, [])
    prod_defs = [d for d in locs if not is_test_path(d["file"])]
    if not prod_defs:
        continue
    all_r = refs.get(name, [])
    prod_r = [r for r in all_r if not r["test"]]
    test_r = [r for r in all_r if r["test"]]
    def_keys = {(d["file"], d["line"]) for d in prod_defs}
    prod_use = [r for r in prod_r if (r["file"], r["line"]) not in def_keys]
    str_prod = [r for r in string_mentions.get(name, []) if not r["test"]]
    routes = decorator_routes.get(name, [])
    is_dunder = name.startswith("__") and name.endswith("__")
    is_keep = name in KEEP_SHARED_SAMPLES

    if prod_use:
        disp = "HAS_PROD_CONSUMER"
    elif routes:
        disp = "FRAMEWORK_ROUTE"
    elif str_prod:
        disp = "DYNAMIC_STRING_REF"
    elif is_dunder:
        disp = "DUNDER"
    elif is_keep:
        disp = "KEEP_SHARED"
    else:
        disp = "ZERO_PROD_REF_CANDIDATE"

    # emit ALL zero-ish + keep + framework for audit; skip boring HAS_PROD to shrink
    if disp == "HAS_PROD_CONSUMER":
        continue
    members.append({
        "name": name,
        "defs": prod_defs[:8],
        "prod_count": len(prod_use),
        "test_count": len(test_r),
        "test_refs": [f"{r['file']}:{r['line']}" for r in test_r[:20]],
        "string_prod_refs": [f"{r['file']}:{r['line']}" for r in str_prod[:10]],
        "framework_routes": routes[:5],
        "disposition_hint": disp,
    })

out = {
    "predicate": (
        "S1 FULL: git ls-files *.py; all prod FunctionDef/ClassDef/Assign(const); "
        "zero non-test Load/Attribute refs → ZERO_PROD_REF_CANDIDATE; "
        "FastAPI/router decorators → FRAMEWORK_ROUTE; string literal mentions → DYNAMIC_STRING_REF; "
        "dunder skip-delete; KEEP_SHARED named. NO sample-name gate for emission of zero-ref set."
    ),
    "tracked_py": len(tracked),
    "zero_candidates": sum(1 for m in members if m["disposition_hint"] == "ZERO_PROD_REF_CANDIDATE"),
    "framework": sum(1 for m in members if m["disposition_hint"] == "FRAMEWORK_ROUTE"),
    "dynamic": sum(1 for m in members if m["disposition_hint"] == "DYNAMIC_STRING_REF"),
    "dunder": sum(1 for m in members if m["disposition_hint"] == "DUNDER"),
    "members": members,
}
(OUTDIR / "s1-full.json").write_text(json.dumps(out, ensure_ascii=False, indent=2), encoding="utf-8")
c = Counter(m["disposition_hint"] for m in members)
print("S1_FULL", dict(c), "emitted", len(members))
print("--- ZERO_PROD_REF_CANDIDATE (first 200) ---")
n = 0
for m in members:
    if m["disposition_hint"] != "ZERO_PROD_REF_CANDIDATE":
        continue
    n += 1
    if n <= 200:
        d0 = m["defs"][0]
        print(f"{m['name']}\t{d0['file']}:{d0['line']}\ttest={m['test_count']}")
print(f"(total zero candidates printed above capped; total={out['zero_candidates']})")
print("--- FRAMEWORK_ROUTE ---")
for m in members:
    if m["disposition_hint"] == "FRAMEWORK_ROUTE":
        d0 = m["defs"][0]
        print(f"{m['name']}\t{d0['file']}:{d0['line']}")
