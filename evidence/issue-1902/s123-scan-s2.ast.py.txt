NO FOCUS symbol filter. Temporary; not a permanent scanner.

OUTDIR.mkdir(parents=True, exist_ok=True)

HELPERS = [

    "map_rescript_option_or_choice",
    "map_rescript_option",

    "apply_dossier_verdicts",

SHAPE_ASSERT_RE = re.compile(
    r"assert\s+(?:batch|mapped|result|out|payload|option|adjudication|rej)\b"
    r"|assert\s+\w+\s*==\s*\{"
    r"|assert\s+isinstance\s*\("
    r"|assert\s+\w+\.(?:keys|values|items)\s*\("
    r"|\.get\(\s*[\"'](?:batch|mapped|options|choices)",
    re.I,
)

def py_tests():
    tracked = subprocess.check_output(["git", "ls-files", "tests/*.py"], text=True).splitlines()
    rows = []
    for path in tracked:
        src = Path(path).read_text(encoding="utf-8")
        try:
            tree = ast.parse(src, filename=path)
        except Exception:
            continue
        for node in ast.walk(tree):
            if not isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
                continue
            if not node.name.startswith("test_"):
                continue
            seg = ast.get_source_segment(src, node) or ""
            orch = sorted({s for s in ORCH if s in seg})
            helpers = sorted({s for s in HELPERS if s in seg})
            raises = "pytest.raises" in seg or "raises(" in seg
            shape = bool(SHAPE_ASSERT_RE.search(seg))
            # mechanical provisional class — fixer overwrites after reading
            if helpers and not orch and (raises or shape or "validate_all" in helpers or any(
                h.startswith("map_rescript") or h.startswith("apply_") or "prewrite" in h
                for h in helpers
            )):
                prov = "INTERNAL_SHAPE_CANDIDATE"
            elif helpers and orch:
                prov = "EXTERNAL_OR_MIXED"
            elif orch:
                prov = "EXTERNAL_REAL_ENTRY"
            elif helpers:
                prov = "HELPER_TOUCH_REVIEW"
            else:
                prov = "EXTERNAL_REAL_ENTRY"  # default: module/unit external to its SUT
            rows.append({
                "kind": "py",
                "file": path,
                "case": node.name,
                "line": node.lineno,
                "orch": orch,
                "helpers": helpers,
                "raises": raises,
                "shape_assert": shape,
                "provisional": prov,
                "contract": None,
                "real_prod_entry": None,
                "external_result": None,
                "existing_tracer": None,
                "duplicate_of": None,
                "disposition": None,
            })
    return rows


IT_RE = re.compile(
    r"""(?:^|\n)([ \t]*)(?:it|test)(?:\.(?:only|skip|todo))?\s*(?:\.each\s*\([^)]*\)\s*)?\(\s*(['"`])((?:\\.|(?!\2).)*)\2""",
    re.M,
)


def web_tests():
    files = subprocess.check_output(["git", "ls-files", "web/"], text=True).splitlines()
    files = [f for f in files if re.search(r"\.(test|spec)\.(ts|tsx|js|jsx)$", f)]
    rows = []
    for path in files:
        src = Path(path).read_text(encoding="utf-8")
        for m in IT_RE.finditer(src):
            title = m.group(3)
            line = src.count("\n", 0, m.start()) + 1
            # grab a window for signals
            win = src[m.start(): m.start() + 2500]
            orch = []
            for s in ("fetch(", "render(", "userEvent", "fireEvent", "screen.", "waitFor",
                       "settleStream", "EventSource", "new Request", "msw", "http.get", "http.post"):
                if s in win:
                    orch.append(s)
            helpers = []
            for s in ("parseSSE", "parse_sse", "JSON.parse", "Number(", "parseFloat", "Number.parse"):
                if s in win:
                    helpers.append(s)
            rows.append({
                "kind": "web",
                "file": path,
                "case": title,
                "line": line,
                "orch": orch,
                "helpers": helpers,
                "raises": "toThrow" in win or "rejects" in win,
                "shape_assert": "toEqual" in win or "toMatchObject" in win,
                "provisional": "EXTERNAL_REAL_ENTRY" if orch else "HELPER_TOUCH_REVIEW" if helpers else "EXTERNAL_REAL_ENTRY",
                "contract": None,
                "real_prod_entry": None,
                "external_result": None,
                "existing_tracer": None,
                "duplicate_of": None,
                "disposition": None,
            })
    return rows


def main():
    py = py_tests()
    web = web_tests()
    rows = py + web
    c = Counter(r["provisional"] for r in rows)
    by_file = defaultdict(list)
    for r in rows:
        by_file[r["file"]].append(r)

    out = {
        "predicate": (
            "S2 FULL: git ls-files tests/*.py ALL test_* + web *.{test,spec}.{ts,tsx} ALL it/test titles. "
            "NO FOCUS symbol whitelist. Provisional labels are mechanical signals only; "
            "fixer fills contract/entry/result/tracer/disposition per case. "
            "Classes: EXTERNAL_REAL_ENTRY | INTERNAL_SHAPE | DUPLICATE | INPUT_ADAPT_OR_DOMAIN_FIRST_WRITE."
        ),
        "py_count": len(py),
        "web_count": len(web),
        "total": len(rows),
        "provisional_counts": dict(c),
        "internal_candidates": [r for r in rows if r["provisional"] == "INTERNAL_SHAPE_CANDIDATE"],
        "helper_touch_review": [r for r in rows if r["provisional"] == "HELPER_TOUCH_REVIEW"],
        "rows": rows,
    }
    (OUTDIR / "s2-full.json").write_text(json.dumps(out, ensure_ascii=False, indent=2), encoding="utf-8")

    # compact TSV of every case
    tsv_lines = ["kind\tfile\tline\tcase\tprovisional\torch\thelpers\traises\tshape"]
    for r in rows:
        tsv_lines.append(
            f"{r['kind']}\t{r['file']}\t{r['line']}\t{r['case']}\t{r['provisional']}\t"
            f"{','.join(r['orch'])}\t{','.join(r['helpers'])}\t{int(r['raises'])}\t{int(r['shape_assert'])}"
        )
    (OUTDIR / "s2-full.tsv").write_text("\n".join(tsv_lines) + "\n", encoding="utf-8")

    print(f"S2_FULL py={len(py)} web={len(web)} total={len(rows)}")
    print("provisional:", dict(c))
    print(f"INTERNAL_SHAPE_CANDIDATE={len(out['internal_candidates'])}")
    print(f"HELPER_TOUCH_REVIEW={len(out['helper_touch_review'])}")
    print("--- INTERNAL_SHAPE_CANDIDATE ---")
    for r in out["internal_candidates"]:
        print(f"{r['file']}:{r['line']}\t{r['case']}\thelpers={r['helpers']}\traises={r['raises']}\tshape={r['shape_assert']}")
    print("--- HELPER_TOUCH_REVIEW (no orch) ---")
    for r in out["helper_touch_review"]:
        print(f"{r['file']}:{r['line']}\t{r['case']}\thelpers={r['helpers']}")


if __name__ == "__main__":
    main()
