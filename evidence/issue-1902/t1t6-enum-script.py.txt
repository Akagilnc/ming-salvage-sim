def iter_files(suffixes, *, roots=None):
    roots = roots or ("ming_sim/", "tests/", "scripts/", "web/src/", "web_app.py")
    for p in ROOT.rglob("*"):
        if not p.is_file() or p.suffix not in suffixes:

        if any(part in SKIP for part in p.parts):
            continue
        rel = p.relative_to(ROOT).as_posix()
        if rel == "web_app.py" or any(rel.startswith(r) for r in roots):
            # skip vendored web tests under node_modules already in SKIP
            if "/node_modules/" in rel:
                continue
            yield p, rel

def read(p: Path) -> str:
    try:
        return p.read_text(encoding="utf-8")
    except Exception:
        return ""

# ── T2: ALL free-output → fault / banner-split candidates (AST+regex, not word list) ──
OUTPUT_NAME_HINT = re.compile(
    r"(stdout|stderr|output|out_text|err_text|prose|banner|text|body|raw|content|"
    r"combined|message|msg|line|chunk|payload_text|cli_out|cli_err|response_text)",
    re.I,
)
FAULT_STATUS_HINT = re.compile(
    r"(llm_connection_error|LLMUnavailable|connection_error|auth(?:entication)?|"
    r"TransportIdle|unavailable|fault|error_code|status_code|raise\s)",
    re.I,
)

class T2Visitor(ast.NodeVisitor):
    def __init__(self, rel, lines):
        self.rel = rel
        self.lines = lines
        self.hits = []

    def _snip(self, node):
        ln = getattr(node, "lineno", 0) or 0
        return {
            "file": self.rel,
            "line": ln,
            "text": (self.lines[ln - 1].strip()[:220] if 0 < ln <= len(self.lines) else ""),
        }

    def visit_Call(self, node: ast.Call):
        # re.search/match/findall/split; str.split/partition; "x" in var
        func = node.func
        name = ""
        if isinstance(func, ast.Attribute):
            name = func.attr
        elif isinstance(func, ast.Name):
            name = func.id
        if name in {"search", "match", "findall", "fullmatch", "sub", "split", "partition",
                    "rpartition", "rsplit"}:
            # any call on re or on an output-ish receiver
            sn = self._snip(node)
            sn["kind"] = f"call:{name}"
            sn["candidate"] = True
            self.hits.append(sn)
        self.generic_visit(node)

    def visit_Compare(self, node: ast.Compare):
        # "needle" in stdout / needle in err
        for op, comp in zip(node.ops, node.comparators):
            if isinstance(op, (ast.In, ast.NotIn)):
                sn = self._snip(node)
                sn["kind"] = "compare_in"
                sn["candidate"] = True
                self.hits.append(sn)
        self.generic_visit(node)

    def visit_Assign(self, node: ast.Assign):
        # status / code / error assignments near output handling — mark assigns of fault-ish names
        for t in node.targets:
            tname = ""
            if isinstance(t, ast.Name):
                tname = t.id
            elif isinstance(t, ast.Attribute):
                tname = t.attr
            if tname and FAULT_STATUS_HINT.search(tname):
                sn = self._snip(node)
                sn["kind"] = f"assign_status:{tname}"
                sn["candidate"] = True
                self.hits.append(sn)
        self.generic_visit(node)

def t2_scan(path: Path, rel: str):
    src = read(path)
    if not src.strip():
        return []
    # cheap prefilter: only files that mention output-ish names OR re. OR split
    if not (OUTPUT_NAME_HINT.search(src) or "re." in src or ".split" in src or " in " in src):
        return []
    try:
        tree = ast.parse(src)
    except SyntaxError:
        return [{"file": rel, "line": 0, "kind": "PARSE_FAIL", "candidate": True, "text": ""}]
    lines = src.splitlines()
    v = T2Visitor(rel, lines)
    v.visit(tree)
    # retain hits whose nearby window mentions output OR fault status (reduces noise but not word-locked)
    out = []
    for h in v.hits:
        ln = h["line"]
        window = "\n".join(lines[max(0, ln - 6): min(len(lines), ln + 5)])
        if OUTPUT_NAME_HINT.search(window) or FAULT_STATUS_HINT.search(window) or h["kind"].startswith("call:"):
            h["window_has_output"] = bool(OUTPUT_NAME_HINT.search(window))
            h["window_has_fault"] = bool(FAULT_STATUS_HINT.search(window))
            out.append(h)
    return out

# ── T3 (same as prior: except → pass/continue/return normal) ──
def t3_scan_py(path: Path, rel: str):
    src = read(path)
    if not src.strip():
        return []
    try:
        tree = ast.parse(src)
    except SyntaxError:
        return [{"file": rel, "line": 0, "kind": "PARSE_FAIL"}]
    lines = src.splitlines()
    out = []
    for node in ast.walk(tree):
        if not isinstance(node, ast.ExceptHandler):
            continue
        kind = None
        for stmt in node.body:
            if isinstance(stmt, ast.Pass):
                kind = "pass"
            elif isinstance(stmt, ast.Continue):
                kind = "continue"
            elif isinstance(stmt, ast.Return):
                v = stmt.value
                if v is None:
                    kind = "return_None_bare"
                elif isinstance(v, ast.Constant) and v.value in (None, "", 0, False):
                    kind = f"return_{v.value!r}"
                elif isinstance(v, (ast.Dict, ast.List, ast.Tuple)) and not (
                    getattr(v, "elts", None) or getattr(v, "keys", None)
                ):
                    kind = "return_empty_collection"
                elif isinstance(v, ast.Name):
                    kind = f"return_name:{v.id}"
                elif isinstance(v, ast.Attribute):
                    kind = "return_attr"
                else:
                    kind = "return_other"
        if not kind:
            continue
        start = max(0, node.lineno - 8)
        end = min(len(lines), (node.end_lineno or node.lineno) + 2)
        window = "\n".join(lines[start:end])
        out.append({
            "file": rel,
            "line": node.lineno,
            "kind": kind,
            "exc_type": ast.unparse(node.type) if node.type else "bare",
            "has_true_cause_log": any(x in window for x in (
                "logger.warning", "logger.exception", "logger.error")),
            "snippet": lines[node.lineno - 1].strip()[:120] if node.lineno <= len(lines) else "",
        })
    return out

ORCH = [

def py_test_funcs(path: Path, rel: str):
    src = read(path)
    if not src.strip():
        return []
    try:
        tree = ast.parse(src)
    except SyntaxError:
        return [{"file": rel, "case": "<PARSE_FAIL>", "line": 0, "lang": "py"}]
    out = []
    for node in tree.body:
        if isinstance(node, ast.FunctionDef) and node.name.startswith("test_"):
            out.append(_py_case(rel, src, node))
        elif isinstance(node, ast.ClassDef):
            for item in node.body:
                if isinstance(item, ast.FunctionDef) and item.name.startswith("test_"):
                    out.append(_py_case(rel, src, item, cls=node.name))
    return out

def _py_case(rel, src, node, cls=None):
    lines = src.splitlines()
    end = node.end_lineno or node.lineno
    body = "\n".join(lines[node.lineno - 1: end])
    orch_hits = [o for o in ORCH if o in body]
    shape = bool(SHAPE.search(body))
    io = bool(IO.search(body))
    # contract fingerprint: asserted call names (rough)
    calls = sorted({m.group(1) for m in re.finditer(r"\b([A-Za-z_][\w\.]{2,})\s*\(", body)})[:40]
    return {
        "file": rel,
        "case": f"{cls}::{node.name}" if cls else node.name,
        "line": node.lineno,
        "lang": "py",
        "orch_hits": orch_hits,
        "has_shape_assert": shape,
        "has_io_or_entry": io,
        "call_fingerprint": calls,
        "body_len": end - node.lineno + 1,
    }

TS_TEST_RE = re.compile(
    r"^\s*(?:it|test|describe)\s*\(\s*['\"`]([^'\"`]+)['\"`]",
    re.M,
)

def ts_test_cases(path: Path, rel: str):
    src = read(path)
    out = []
    for m in TS_TEST_RE.finditer(src):
        # take a window of ~40 lines after match for signals
        start = src[:m.start()].count("\n") + 1
        window = "\n".join(src.splitlines()[start - 1: start + 50])
        out.append({
            "file": rel,
            "case": m.group(1)[:160],
            "line": start,
            "lang": "ts",
            "orch_hits": [o for o in ORCH if o in window],
            "has_shape_assert": bool(re.search(r"toEqual|toStrictEqual|toMatchObject|typeof|instanceof", window)),
            "has_io_or_entry": bool(re.search(r"fetch\(|msw|http|render\(|fireEvent|userEvent|api\.", window, re.I)),
            "call_fingerprint": [],
            "body_len": 50,
        })
    return out

def classify_t5(case):
    """Mechanical pre-class only; semantic disposition is separate."""
    if case["has_io_or_entry"] or case["orch_hits"]:
        if case["has_shape_assert"] and not case["has_io_or_entry"]:
            return "CANDIDATE_MIXED_SHAPE_WITH_ORCH_NAME"
        return "CANDIDATE_EXTERNAL_OR_ENTRY"
    if case["has_shape_assert"]:
        return "CANDIDATE_INTERNAL_SHAPE"
    # no io, no shape — still a candidate for duplicate-contract grouping via fingerprint
    return "CANDIDATE_BEHAVIOR_OR_UNKNOWN"

# ── T6: ALL production symbols + ref counts (non-test) ──
def collect_prod_defs():
    defs = []  # {qual, file, line, kind, name}
    for p, rel in iter_files({".py"}, roots=("ming_sim/", "scripts/", "web_app.py")):
        if rel.startswith("tests/"):
            continue
        src = read(p)
        if not src.strip():
            continue
        try:
            tree = ast.parse(src)
        except SyntaxError:
            continue
        for node in tree.body:
            if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
                defs.append({"name": node.name, "file": rel, "line": node.lineno, "kind": "func"})
            elif isinstance(node, ast.ClassDef):
                defs.append({"name": node.name, "file": rel, "line": node.lineno, "kind": "class"})
                for item in node.body:
                    if isinstance(item, (ast.FunctionDef, ast.AsyncFunctionDef)):
                        # methods — still symbols for retirement wrappers
                        if not item.name.startswith("__"):
                            defs.append({
                                "name": f"{node.name}.{item.name}",
                                "bare": item.name,
                                "file": rel, "line": item.lineno, "kind": "method",
                            })
    return defs

def count_refs(name: str, *, bare: str | None = None):
    """Count production vs test references via rg (literal)."""
    import subprocess
    needle = bare or name.split(".")[-1]
    # skip very short / common names
    if len(needle) < 3 or needle in {
        "get", "set", "add", "run", "call", "open", "close", "read", "write", "main",
        "data", "info", "load", "save", "init", "post", "put",
    }:
        return {"prod": -1, "test": -1, "skipped": True}
    try:
        r = subprocess.run(
            ["rg", "-l", "--glob", "*.py", "--glob", "!**/node_modules/**", needle, str(ROOT)],
            capture_output=True, text=True, timeout=60,
        )
    except Exception as e:
        return {"prod": -1, "test": -1, "error": str(e)}
    files = [ln.strip() for ln in r.stdout.splitlines() if ln.strip()]
    prod, test = [], []
    for f in files:
        rel = str(Path(f).resolve()).replace(str(ROOT) + "/", "")
        if "/node_modules/" in rel or rel.startswith("evidence/"):
            continue
        if rel.startswith("tests/") or "/tests/" in rel:
            test.append(rel)
        elif rel.endswith(".py"):
            prod.append(rel)
    return {"prod": len(prod), "test": len(test), "prod_files": prod[:8], "test_files": test[:8]}

def main():
    results = {
        "predicates": {
            "T1": "拆除显式及隐式强杀/硬期限；合法静默保留；终止缺口如实报告；不另造监督层。",
            "T2": "删除从自由输出词面判故障、切分呈现横幅的机制及钉误判测试；用原生状态/结构化契约。枚举一切输出变量分支/正则/切分及结果状态候选，不以指定词穷尽。",
            "T3": "代码故障响亮；确证安全旁路保留真因日志；不给所有 catch 加统一框架。",
            "T4": "合并恢复生命周期，删入口编排副本；保留输入/呈现/宿主适配。snapshot 失败不得伪造差异，亦不声称跳过 diff 仍保副作用回滚。",
            "T5": "全仓现存 Py/TS 测试函数按真实入口/调用/断言契约归类与重复契约分组；处置纯内部与重复证明；不按行数/操作符盲删。",
            "T6": "全生产符号引用/可达入口（非测试）候选 ∧ 当前消费者清退授权内退役类子树/独占证明；撤销底座时点豁免；保留真实共享能力。",
        },
        "T1": [],
        "T2": [],
        "T3": [],
        "T4": [],
        "T5": [],
        "T6_prod_symbols": [],
    }

    for p, rel in iter_files({".py", ".ts", ".tsx"}):
        text = read(p)
        if not text:
            continue
        for i, line in enumerate(text.splitlines(), 1):
            if T1_RE.search(line):
                results["T1"].append({"file": rel, "line": i, "text": line.strip()[:200],
                                      "scope": "test" if rel.startswith("tests/") else "prod"})
            if T4_RE.search(line):
                results["T4"].append({"file": rel, "line": i, "text": line.strip()[:200]})

        # T2 AST on py; line-level split/re for ts
        if rel.endswith(".py"):
            results["T2"].extend(t2_scan(p, rel))
            if not rel.startswith("tests/"):
                results["T3"].extend(t3_scan_py(p, rel))
        else:
            for i, line in enumerate(text.splitlines(), 1):
                if re.search(r"\.(?:split|match|search|partition)\s*\(|\bin\b.*(?:error|auth|banner)", line, re.I):
                    if OUTPUT_NAME_HINT.search(line) or FAULT_STATUS_HINT.search(line):
                        results["T2"].append({"file": rel, "line": i, "kind": "ts_line",
                                             "candidate": True, "text": line.strip()[:200]})

    # T5 full test universe
    for p, rel in iter_files({".py"}, roots=("tests/",)):
        if not rel.startswith("tests/"):
            continue
        for case in py_test_funcs(p, rel):
            case["preclass"] = classify_t5(case)
            results["T5"].append(case)
    for p, rel in iter_files({".ts", ".tsx"}, roots=("web/src/",)):
        if not (rel.endswith(".test.ts") or rel.endswith(".test.tsx") or rel.endswith(".spec.ts")):
            continue
        for case in ts_test_cases(p, rel):
            case["preclass"] = classify_t5(case)
            results["T5"].append(case)

    # duplicate-contract groups: same file-prefix + same call_fingerprint top-10
    fp_groups = defaultdict(list)
    for c in results["T5"]:
        fp = tuple(c.get("call_fingerprint") or [])[:12]
        if not fp:
            continue
        key = (c["preclass"], fp)
        fp_groups[key].append(f"{c['file']}::{c['case']}")
    dup_groups = [
        {"preclass": k[0], "fingerprint": list(k[1]), "count": len(v), "members": v[:30]}
        for k, v in fp_groups.items() if len(v) >= 3
    ]
    dup_groups.sort(key=lambda x: -x["count"])

    # T6: production symbols (name-level). Full ref count is expensive — do for
    # (a) auth-retire-hint names, (b) zero-appearance shortlist via name in file only once.
    defs = collect_prod_defs()
    # Always include auth-hint symbols + top-level funcs/classes (not every method)
    t6_focus = []
    for d in defs:
        name = d["name"]
        bare = d.get("bare") or name.split(".")[-1]
        auth = bool(T6_AUTH_RETIRE_HINTS.search(name) or T6_AUTH_RETIRE_HINTS.search(d["file"]))
        if d["kind"] == "method" and not auth:
            continue  # methods only if auth-hint
        t6_focus.append({**d, "auth_retire_hint": auth, "bare": bare})

    # ref-count auth-hint first, then a sample of top-level with hint file paths related to entry
    for d in t6_focus:
        if d["auth_retire_hint"] or d["kind"] in {"func", "class"}:
            # only full-count auth OR names matching retire vocabulary in body of s123 KEEP list later
            if d["auth_retire_hint"]:
                d["refs"] = count_refs(d["name"], bare=d["bare"])
            else:
                d["refs"] = None  # filled selectively below
        results["T6_prod_symbols"].append(d)

    # Also load KEEP_NEVER_WIRED names and re-count each (authorized filter applied in disposition)
    knw = []
    disp = ROOT / "evidence/issue-1902/s123-s1-disposition.md"
    if disp.exists():
        for line in disp.read_text().splitlines():
            m = re.match(r"\|\s*`([^`]+)`\s*\|\s*\*\*KEEP\*\*\s*\|\s*KEEP_NEVER_WIRED", line)
            if m:
                knw.append(m.group(1))
    knw_audit = []
    for name in knw:
        refs = count_refs(name)
        auth = bool(T6_AUTH_RETIRE_HINTS.search(name))
        knw_audit.append({"name": name, "refs": refs, "auth_retire_hint": auth})

    summary = {
        "T1": len(results["T1"]),
        "T2": len(results["T2"]),
        "T3": len(results["T3"]),
        "T4": len(results["T4"]),
        "T5": len(results["T5"]),
        "T5_preclass": dict(Counter(c["preclass"] for c in results["T5"])),
        "T5_dup_groups_ge3": len(dup_groups),
        "T6_prod_symbols_focused": len(results["T6_prod_symbols"]),
        "T6_auth_hint_symbols": sum(1 for d in results["T6_prod_symbols"] if d.get("auth_retire_hint")),
        "T6_KEEP_NEVER_WIRED": len(knw_audit),
        "T6_KEEP_NEVER_WIRED_auth_hint": sum(1 for x in knw_audit if x["auth_retire_hint"]),
    }
    out = {
        "summary": summary,
        "results": results,
        "T5_dup_groups": dup_groups[:80],
        "T6_KEEP_NEVER_WIRED_audit": knw_audit,
    }
    (OUT / "raw_full.json").write_text(json.dumps(out, ensure_ascii=False, indent=2), encoding="utf-8")
    (OUT / "summary.json").write_text(json.dumps(summary, ensure_ascii=False, indent=2), encoding="utf-8")
    print(json.dumps(summary, ensure_ascii=False, indent=2))
    print("wrote", OUT / "raw_full.json")

if __name__ == "__main__":
    main()
