"""Full-repo AST candidate enumeration for J18/J6 class shapes (one-shot)."""
from __future__ import annotations
import ast, json, re
from pathlib import Path

ROOT = Path('.').resolve()
OUT = ROOT / 'evidence/1900-final-class-repair'
SKIP_DIRS = {'.git', 'evidence', '.baseline', 'node_modules', 'dist', '__pycache__', '.venv', 'venv', 'baseline-aside'}

J18_NAME_PATTERNS = [
    r'escort', r'护送', r'护行', r'暗护', r'押解',
    r'grant_route', r'dossier_escort', r'escort_sources', r'escort_pending',
    r'is_grant_allocation', r'_write_dossier_payload', r'_GRANT_ESCORT',
    r'reader_may_cite_escort', r'escort_identity', r'covert_participant',
    r'软判提案', r'grant_arrival_bounds',
]

def should_skip(p: Path) -> bool:
    return bool(set(p.parts) & SKIP_DIRS)

j18_ast, j6_ast = [], []

class Visitor(ast.NodeVisitor):
    def __init__(self, path, kind):
        self.path, self.kind = path, kind
    def visit_FunctionDef(self, node):
        name = node.name
        if self.kind == 'prod':
            for pat in J18_NAME_PATTERNS:
                if re.search(pat, name, re.I):
                    j18_ast.append({
                        'file': str(self.path), 'line': node.lineno, 'kind': 'def',
                        'name': name, 'pattern': pat,
                    })
                    break
        if self.kind == 'test' and name.startswith('test_'):
            src = ast.get_source_segment(open(self.path, encoding='utf-8').read(), node) or ''
            hits = []
            if 'monkeypatch.setattr' in src or 'monkeypatch.setitem' in src:
                hits.append('monkeypatch')
            if re.search(r'assert\s+\w+\s*[<>]\s*\w+', src):
                hits.append('direction_compare')
            if '_has_meta_flag' in src or '__leverage' in src:
                hits.append('marker')
            if 'assert_called' in src or 'call_count' in src or 'seen.get' in src:
                hits.append('call_oracle')
            if re.search(r'["\']OperationalError["\']', src) or 'in message' in src or 'in msg' in src:
                hits.append('prose_lock')
            if 'routed:' in src or "routed'" in src:
                hits.append('routed_oracle')
            if hits:
                j6_ast.append({
                    'file': str(self.path), 'line': node.lineno, 'name': name, 'hits': hits,
                })
        self.generic_visit(node)
    visit_AsyncFunctionDef = visit_FunctionDef

for p in ROOT.rglob('*.py'):
    if should_skip(p):
        continue
    rel = p.relative_to(ROOT)
    kind = 'test' if str(rel).startswith('tests/') or 'test_' in p.name else 'prod'
    if kind == 'prod' and not (str(rel).startswith('ming_sim/') or rel.name == 'web_app.py'):
        continue
    try:
        tree = ast.parse(p.read_text(encoding='utf-8'), filename=str(p))
    except Exception:
        continue
    Visitor(rel, kind).visit(tree)

print('J18_AST_DEFS', len(j18_ast))
print('J6_AST_TESTS', len(j6_ast))
OUT.mkdir(parents=True, exist_ok=True)

"""J6 fine AST: stub targets / call·marker oracle / prose / direction / fixed translate / helper."""
from __future__ import annotations
import ast, json, re
from pathlib import Path
from collections import defaultdict

ROOT = Path('.').resolve()
OUT = ROOT / 'evidence/1900-final-class-repair'
SKIP = {'.git','evidence','.baseline','node_modules','dist','__pycache__','.venv','venv','baseline-aside','.pytest_cache'}

def skip(p: Path) -> bool:
    return bool(set(p.parts) & SKIP)

J6_CHECKS = [
    ('stub_under_test', re.compile(r'monkeypatch\.setattr\(\s*(?:[^\n]{0,80}),\s*[\'\"]([^\'\"]+)[\'\"]')),
    ('call_oracle', re.compile(r'(call_count|assert_called|seen\.get\(|routed:|spy_calls|AUTO_CLOSE|track_auto_close)')),
    ('marker_oracle', re.compile(r'(_has_meta_flag|__leverage_migrated|__meta_flag)')),
    ('prose_lock_diag', re.compile(r'(OperationalError|[\'\"][\u4e00-\u9fff]{4,}[\'\"]\s+in\s+\w+)')),
    ('direction', re.compile(r'assert\s+(\w+)\s*([<>])\s*(\w+)\b')),
    ('fixed_translate', re.compile(r'translate_fn\s*=\s*lambda')),
    ('helper_name', re.compile(r'^test_.*(helper|internal|_private_|_format_|_build_).*', re.I)),
    ('retire_prove', re.compile(r'(is_retired|already_retired|退役证明|no longer exists|AttributeError.*getattr)')),
]

j6 = []
for p in (ROOT/'tests').rglob('*.py'):
    if skip(p):
        continue
    rel = str(p.relative_to(ROOT))
    text = p.read_text(encoding='utf-8')
    try:
        tree = ast.parse(text, filename=rel)
    except SyntaxError:
        continue
    stack = list(tree.body)
    while stack:
        node = stack.pop()
        if isinstance(node, ast.ClassDef):
            stack.extend(node.body)
            continue
        if not isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            continue
        if not node.name.startswith('test_'):
            continue
        src = ast.get_source_segment(text, node) or ''
        hits = []
        for tag, rx in J6_CHECKS:
            if tag == 'helper_name':
                if rx.search(node.name):
                    hits.append(tag)
            elif tag == 'stub_under_test':
                for m in rx.finditer(src):
                    hits.append(f'stub:{m.group(1)}')
            elif tag == 'direction':
                for m in rx.finditer(src):
                    left, op, right = m.group(1), m.group(2), m.group(3)
                    if right in ('0','1','None') or left in ('len',):
                        continue
                    if re.match(r'^\d+$', right) or re.match(r'^\d+$', left):
                        continue
                    hits.append(f'direction:{left}{op}{right}')
            elif rx.search(src):
                hits.append(tag)
        if hits:
            j6.append({'file': rel, 'line': node.lineno, 'name': node.name, 'hits': sorted(set(hits))})

web_j6 = []
for p in (ROOT/'web').rglob('*'):
    if skip(p) or p.suffix not in ('.ts', '.tsx'):
        continue
    if not (p.name.endswith('.test.ts') or p.name.endswith('.test.tsx') or p.name.endswith('.spec.ts') or p.name.endswith('.spec.tsx') or '__tests__' in p.parts):
        continue
    rel = str(p.relative_to(ROOT))
    try:
        text = p.read_text(encoding='utf-8')
    except Exception:
        continue
    for i, line in enumerate(text.splitlines(), 1):
        if re.search(r'vi\.spyOn|toHaveBeenCalled|mockImplementation|jest\.spyOn', line):
            web_j6.append({'file': rel, 'line': i, 'text': line.strip()[:160]})

print('J6 python candidates', len(j6))
print('J6 web mock lines', len(web_j6))
