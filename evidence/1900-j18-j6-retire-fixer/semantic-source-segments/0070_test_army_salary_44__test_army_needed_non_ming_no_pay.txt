def test_army_needed_non_ming_no_pay(read_game):
    # 非明军（owner_power != ming）不强加饷需（叛军/外族不吃明国库）。
    db, state, _ = read_game
    row = db.conn.execute(
        "SELECT * FROM armies WHERE owner_power!='ming' LIMIT 1").fetchone()
    if row is None:
        pytest.skip("无非明军")
    assert army_needed(row) == 0
