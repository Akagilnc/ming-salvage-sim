def test_capture_manual_directive_keeps_real_person_participant(game, monkeypatch):
    """着毕自严… 仍照常抽人（名册可解析人名保留）。"""
    ids = _capture_ids(
        game, monkeypatch,
        text="着毕自严核清太仓实存，边饷优先",
        roster=[{"character_id": "毕自严", "tier": "主办", "role": "核太仓"}],
    )
    assert ids == ["毕自严"]
