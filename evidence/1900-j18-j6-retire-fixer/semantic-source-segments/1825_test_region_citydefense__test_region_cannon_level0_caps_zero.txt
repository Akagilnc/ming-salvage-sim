def test_region_cannon_level0_caps_zero(game):
    """level 0 的地方(游牧/孤岛)城防炮上限 0。"""
    db, state, _ = game
    db.apply_region_cannon(state, "mongol_chahar", 50)
    assert db.conn.execute("SELECT cannon FROM regions WHERE id='mongol_chahar'").fetchone()[0] == 0
