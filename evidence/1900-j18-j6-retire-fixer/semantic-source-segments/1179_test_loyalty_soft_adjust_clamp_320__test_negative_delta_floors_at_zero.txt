def test_negative_delta_floors_at_zero(game):
    """①c 触底：loyalty=5 喂 -50 → DB=0。"""
    db, state, _ = game
    _set_loyalty(db, loyalty=5)

    _apply(db, state, {"loyalty": -50})

    assert _loyalty(db) == 0
