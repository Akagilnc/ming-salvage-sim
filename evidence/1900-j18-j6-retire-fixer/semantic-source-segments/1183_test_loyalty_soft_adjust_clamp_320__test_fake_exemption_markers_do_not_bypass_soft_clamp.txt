def test_fake_exemption_markers_do_not_bypass_soft_clamp(game, kwargs):
    """⑤ 伪豁免标记一律仍受 ±15。"""
    db, state, _ = game
    _set_loyalty(db, loyalty=40)

    _apply(
        db,
        state,
        kwargs["delta"],
        event=kwargs.get("event"),
        origin_ref=kwargs.get("origin_ref", ""),
    )

    assert _loyalty(db) == 55
