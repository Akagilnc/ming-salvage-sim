def test_mutiny_stays_latched_while_loyalty_recovers_below_40(game, fiscal_path):
    db, state, _ = game
    _setup(db, fiscal_path, loyalty=20, arrears=0, latched=1)

    trajectory = [_tick(db, state) for _ in range(3)]
    assert trajectory == [
        (25, 1, "哗变"), (30, 1, "哗变"), (35, 1, "哗变")
    ]
