def test_latched_denies_dispatch_armament_status_and_positive_manpower(game):
    db, state, _ = game
    _configure(db, "legacy")
    _set(db, "legacy", loyalty=30, arrears=5, latched=1, mutiny_count=1)
    before = _snapshot(db)

    db.apply_army_deltas(
        state,
        _event(),
        None,
        "测试",
        {
            ARMY: {
                "station": "新驻地",
                "commander": "新统帅",
                "status": "已整编",
                "equipment": 5,
                "firearm_equipment": 5,
                "cannon_equipment": 1,
                "training": 5,
                "troop_type": "新兵种",
                "morale": 5,
                "supply": 5,
                "manpower": 100,
            }
        },
    )

    after = _snapshot(db)
    assert after == before
