def test_capture_manual_directive_keeps_surname_title_aliases(
    game, monkeypatch, alias, canonical,
):
    """带姓称谓别名不得当机构丢；走 canon 留人名（回归别名面，不只钉 aliases[0] 本名）。"""
    ids = _capture_ids(
        game, monkeypatch,
        text=f"着{alias}核办边饷",
        roster=[{"character_id": alias, "tier": "主办"}],
    )
    assert ids == [canonical]
