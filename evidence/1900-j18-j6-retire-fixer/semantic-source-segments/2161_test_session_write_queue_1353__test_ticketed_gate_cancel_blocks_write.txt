def test_ticketed_gate_cancel_blocks_write():
    """生产 seam：TicketedWriteGate 见取消不进写临界区。"""
    q = SessionWriteQueue()
    t = q.claim(key=("turn", 7))
    assert t is not None
    q.cancel(t)
    wrote = {"n": 0}
    gate = q.ticketed_gate(t)
    with pytest.raises(TicketCancelled):
        with gate:
            wrote["n"] += 1
    assert wrote["n"] == 0
