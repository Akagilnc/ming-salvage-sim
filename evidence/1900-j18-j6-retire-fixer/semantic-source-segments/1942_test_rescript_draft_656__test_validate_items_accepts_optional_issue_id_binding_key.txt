def test_validate_items_accepts_optional_issue_id_binding_key(draft_ingress):
    """issue_id 是唯一豁免的可选绑定键，白名单收窄不得误伤既有绑定路。"""
    item = _legal_item()
    item["issue_id"] = 42
    drafts = draft_ingress({"items": [item]}, {42})
    assert drafts[0]["event_id"] == "issue:42"
