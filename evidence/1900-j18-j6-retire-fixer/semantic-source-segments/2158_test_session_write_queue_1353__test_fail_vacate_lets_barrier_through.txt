def test_fail_vacate_lets_barrier_through():
    """失败钉：腿失败票据空放行，屏障不卡死。"""
    q = SessionWriteQueue()
    t = q.claim(key=("extract", 9))
    assert t is not None
    q.vacate(t)
    order: list[str] = []
    q.barrier(lambda: order.append("barrier") or None)
    assert order == ["barrier"]
    assert q.inflight_count() == 0
