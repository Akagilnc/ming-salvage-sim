## 枚举命令（全仓 git ls-files；输出到 /tmp，不写仓库）

成员/例外真源：members-exceptions.txt
候选明细不入库；需要时执行下列命令写出 `/tmp/1897-r1r2-candidates-ast.txt`。

```bash
cd /Users/akagilnc/WorkSpace/Ming_LLM-1897-w5 && python3 <<'PY'
from __future__ import annotations
import ast, re, subprocess
from pathlib import Path
ROOT = Path('.').resolve()
OUT = Path('/tmp/1897-r1r2-candidates-ast.txt')
METHODS = {
    'strip','lstrip','rstrip','replace','split','rsplit',
    'partition','rpartition','removeprefix','removesuffix',
}
RE_FUNCS = {'sub','split','subn','findall','finditer'}
SRC_SUFFIX = {'.py','.pyi','.ts','.tsx','.js','.jsx','.mjs','.cjs',
              '.rs','.go','.java','.kt','.swift','.rb','.php','.cs'}
files = [b.decode() for b in subprocess.check_output(['git','ls-files','-z']).split(b'\0') if b]
paths = [p for p in files if Path(p).suffix in SRC_SUFFIX]
py_paths = [p for p in paths if p.endswith('.py')]
non_py = [p for p in paths if not p.endswith('.py')]
hits = []
class V(ast.NodeVisitor):
    def __init__(self, path): self.path=path; self.stack=[]
    def _q(self): return '.'.join(self.stack) or '<module>'
    def visit_ClassDef(self,n):
        self.stack.append(n.name); self.generic_visit(n); self.stack.pop()
    def visit_FunctionDef(self,n):
        self.stack.append(n.name); self.generic_visit(n); self.stack.pop()
    visit_AsyncFunctionDef = visit_FunctionDef
    def _add(self,n,kind,detail,preview=''):
        hits.append(f"{self.path}:{getattr(n,'lineno',None)}\t{self._q()}\t{kind}\t{detail}\t{preview[:160]}")
    def visit_Call(self,n):
        f=n.func
        if isinstance(f, ast.Attribute) and f.attr in METHODS:
            self._add(n,'call',f.attr,ast.dump(n,include_attributes=False))
        if isinstance(f, ast.Attribute) and isinstance(f.value, ast.Name) and f.value.id=='re' and f.attr in RE_FUNCS:
            self._add(n,'re',f.attr,ast.dump(n,include_attributes=False))
        self.generic_visit(n)
    def visit_Subscript(self,n):
        sl=n.slice
        if isinstance(sl, ast.Slice) and sl.upper is not None and sl.step is None:
            upper = sl.upper.value if isinstance(sl.upper, ast.Constant) and isinstance(sl.upper.value, int) else '?'
            self._add(n,'slice',f'[:{upper}]',ast.dump(n,include_attributes=False))
        self.generic_visit(n)
    def visit_BinOp(self,n):
        if isinstance(n.op, ast.Add):
            dump=ast.dump(n,include_attributes=False)
            if 'Constant(value=' in dump:
                self._add(n,'concat','+',dump)
        self.generic_visit(n)
    def visit_JoinedStr(self,n):
        has_fmt=any(isinstance(v,ast.FormattedValue) for v in n.values)
        has_const=any(isinstance(v,ast.Constant) and isinstance(v.value,str) and v.value for v in n.values)
        if has_fmt and has_const:
            self._add(n,'fstring_assemble','f-string',ast.dump(n,include_attributes=False))
        self.generic_visit(n)
for rel in sorted(py_paths):
    tree=ast.parse((ROOT/rel).read_text(encoding='utf-8',errors='replace'), filename=rel)
    V(rel).visit(tree)
pat=re.compile(r'\.(strip|lstrip|rstrip|replace|split)\(|re\.sub\(|\[:\s*\d+\]')
rg=[]
for rel in sorted(non_py):
    text=(ROOT/rel).read_text(encoding='utf-8',errors='replace')
    for i,line in enumerate(text.splitlines(),1):
        if pat.search(line):
            rg.append(f'{rel}:{i}\trg_line\t{line.strip()[:160]}')
with OUT.open('w',encoding='utf-8') as f:
    f.write(f'# py_files={len(py_paths)} py_hits={len(hits)} non_py_hits={len(rg)}\n')
    for h in hits: f.write(h+'\n')
    for x in rg: f.write(x+'\n')
print(OUT, 'py_hits', len(hits), 'non_py', len(rg))
PY
```
