  按真实历史消费者及当前活入口追完整支持闭包；现役表和共享读写不等于其旧档迁移合法。

Class B boundary (末判 fix.classes[1]):

Predicates cover the class definition text — not sample-name / startswith / hasattr
whitelists. No fixed seed thresholds, ref-count gates, history lower bounds, or
same-file live-code whitelists as class narrowing. Seeds are chase starts only.
Consumer site lists are never truncated in TSV (full ;-joined path:line).

def iter_files(bases, suffixes={".py", ".ts", ".tsx", ".js", ".jsx", ".md", ".json"}):
    for base in bases:
        if not base.exists():
            continue
        for p in base.rglob("*"):
            if not p.is_file():
                continue
            if p.suffix not in suffixes:
                continue
            if any(x in p.parts for x in ("__pycache__", "node_modules", "dist", ".venv")):
                continue
            yield p


def rel(p: Path) -> str:
    return str(p.relative_to(ROOT))


def parse_py(path: Path):
    try:
        return ast.parse(path.read_text(encoding="utf-8"), filename=str(path))
    except (SyntaxError, UnicodeDecodeError):
        return None


defs = {}  # qual -> meta
loads = defaultdict(list)  # bare name -> sites
attr_loads = defaultdict(list)
resources = []  # content resources touching old gazette / simulator

for path in iter_files(CODE_DIRS, {".py"}):
    tree = parse_py(path)
    if tree is None:
        continue
    r = rel(path)
    in_test = r.startswith("tests/")
    for node in ast.walk(tree):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
            q = f"{r}::{node.name}"
            defs[q] = {
                "path": r,
                "name": node.name,
                "lineno": node.lineno,
                "end": getattr(node, "end_lineno", node.lineno),
                "in_test": in_test,
                "kind": type(node).__name__,
            }
        if isinstance(node, ast.Name) and isinstance(node.ctx, ast.Load):
            loads[node.id].append({"path": r, "lineno": node.lineno})
        if isinstance(node, ast.Attribute) and isinstance(node.ctx, ast.Load):
            attr_loads[node.attr].append({"path": r, "lineno": node.lineno})

# Content / resource inventory for old-compat assets
for path in iter_files([ROOT / "content"], {".md", ".json", ".txt"}):
    r = rel(path)
    name = path.name.lower()
    if any(k in name for k in ("opening_gazette", "simulator", "legacy_board", "turn_summary")):
        resources.append({"path": r, "kind": "content_resource", "name": path.name})


def sites_for(name: str):
    return list(loads.get(name, [])) + list(attr_loads.get(name, []))


def external_sites(name: str, def_path: str, lo: int, hi: int):
    out = []
    for s in sites_for(name):
        if s["path"] == def_path and lo <= s["lineno"] <= hi:
            continue
        if s["path"] == def_path and abs(s["lineno"] - lo) <= 2:
            continue
        out.append(s)
    return out


def fmt_sites(sites):
    return ";".join(f"{s['path']}:{s['lineno']}" for s in sites)


def prod_sites(sites):
    return [s for s in sites if not s["path"].startswith("tests/")]


def test_sites(sites):
    return [s for s in sites if s["path"].startswith("tests/")]

            "test_sites": fmt_sites(tc or []),
            "extra": extra,
        }
    )

def add_b(member, kind, disposition, reason, lineno=0, flags=""):
    key = f"{member}|{kind}|{lineno}"

# Full-repo (tests + web) test-function / file scan — not limited to startswith/hasattr/army_pay
ENTRY_RE = re.compile(
    r"apply_fixed_period_flows|apply_score_extraction|GameDB\(|GameSession\(|"
    r"run_player_month_chain|assemble_structured_decree|gate_llm_config|"
    r"create_rescript|follow_draft|submit_|/api/|web_game|admit_audience|"
    r"load_region_content|canonicalize_location|create_decree_dossier|"
    r"restore_pay_order|apply_dossier|list_rescript|save_runtime_llm|"
    r"pytest\.raises"
)

for path in iter_files(TEST_DIRS + [d for d in WEB_DIRS if d.exists()], {".py", ".ts", ".tsx", ".js"}):
    r = rel(path)
    try:
        text = path.read_text(encoding="utf-8")
    except UnicodeDecodeError:
        continue
    if path.suffix != ".py" or not r.startswith("tests/"):
        # web non-py: line-level wording/internal probes
        for i, line in enumerate(text.splitlines(), 1):
            if re.search(r"inspect\.getsource|assert\s+not\s+hasattr|toMatch\(\s*['\"]\[", line):
                add_b(f"{r}:{i}", "web_wording_or_internal", "CANDIDATE",
                      "web 源码/内部/措辞锁候选", i, line.strip()[:160])
        continue
    tree = parse_py(path)
    if tree is None:
        continue
    lines = text.splitlines()
    # duplicate detection later via body hash
    bodies = {}
    for node in ast.walk(tree):
        if not isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            continue
        if not node.name.startswith("test_"):
            continue
        lo, hi = node.lineno, getattr(node, "end_lineno", node.lineno)
        body = "\n".join(lines[lo - 1 : hi])
        member = f"{r}::{node.name}"
        flags = []
        # source lock
        if re.search(r"inspect\.getsource|\.getsource\s*\(", body):
            flags.append("source_lock_getsource")
        # reading implementation .py as assertion oracle (not durable artifact)
        if re.search(
            r"(Path\([^\)]*ming_sim[^\)]*\)|open\([^\)]*ming_sim/[^\)]*\.py)."
            r"{0,80}read_text|"
            r"read_text\([^\)]*\).{0,40}ming_sim/",
            body,
            re.S,
        ):
            flags.append("source_lock_read_impl")
        # exception match literal wording
        if re.search(r"pytest\.raises\([^)]*match\s*=\s*['\"][^'\"]+['\"]", body):
            flags.append("exception_match_literal")
        # exception match variable — may be same-value pass-through
        elif re.search(r"pytest\.raises\([^)]*match\s*=", body):
            flags.append("exception_match_param")
        # presentation / log prefix wording
        if re.search(r"\.startswith\(\s*['\"]\[", body) or re.search(
            r"\.startswith\(\s*['\"]#\d+", body
        ):
            flags.append("log_or_issue_prefix_wording")
        # any assert startswith — candidate; disposition distinguishes structural id vs wording
        if re.search(r"assert\s+.+\.startswith\s*\(", body):
            flags.append("assert_startswith")
        # internal attr absence
        if re.search(r"assert\s+not\s+hasattr\s*\(", body):
            flags.append("internal_attr_absence")
        # hasattr on SUT private internals (not fixture probes)
        if re.search(r"hasattr\s*\(\s*\w+\s*,\s*['\"]_[A-Za-z]", body):
            flags.append("hasattr_private_internal")
        # weakened same-value: notes/defaults startswith fixed ticket prefix
        if re.search(r"(notes|default|说明).{0,40}startswith\(|startswith\(\s*['\"]#\d+", body):
            flags.append("weakened_same_value_prefix")
        # pure formula helper without real entry
        if re.search(r"from ming_sim\.army_pay import army_pay_morale_delta", body) and not ENTRY_RE.search(body):
            flags.append("pure_formula_helper")
        # broader pure-private-helper: only asserts on _private returns, no entry
        priv_calls = re.findall(r"\b_[a-zA-Z][a-zA-Z0-9_]*\s*\(", body)
        if (
            priv_calls
            and not ENTRY_RE.search(body)
            and "game" not in {a.arg for a in node.args.args}
            and re.search(r"assert\s+", body)
            and not re.search(r"GameDB|GameSession|client\.|fastapi|TestClient", body)
        ):
            # only flag if body imports production private and no db fixture
            if re.search(r"from ming_sim\.[^\n]+ import[^\n]*_", body):
                flags.append("helper_internal_only")
        # duplicate constant subset + useless game
        if "game" in {a.arg for a in node.args.args} and "_ = game" in body and re.search(
            r"RESCRIPT_EMITTED_DOSSIER_ACTION_TYPES\s*<=|DOSSIER_ACTION_TYPES", body
        ):
            flags.append("duplicate_constant_subset_useless_fixture")
        # prompt/log wording equality on long Chinese literals
        if re.search(r"assert\s+['\"][\u4e00-\u9fff]{6,}['\"]\s+in\s+\w+", body):
            flags.append("chinese_literal_in_wording")

        if not flags:
            continue
        bodies[member] = body
        add_b(
            member,
            ",".join(flags),
            "CANDIDATE",
            "全仓测试措辞/内部/弱化/纯helper/重复候选；须按契约分类处置",
            lo,
            ";".join(flags),
        )

# duplicate bodies (exact) among flagged and general tests in same file already covered;
# cross-file exact duplicate of flagged bodies
from hashlib import sha1

hash_to = defaultdict(list)
for path in iter_files(TEST_DIRS, {".py"}):
    tree = parse_py(path)
    if tree is None:
        continue
    lines = path.read_text(encoding="utf-8").splitlines()
    r = rel(path)
    for node in ast.walk(tree):
        if not isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            continue
        if not node.name.startswith("test_"):
            continue
        body = "\n".join(lines[node.lineno - 1 : getattr(node, "end_lineno", node.lineno)])
        # normalize def line name away for duplicate detection
        norm = re.sub(r"^def test_\w+", "def test_X", body)
        h = sha1(norm.encode()).hexdigest()[:16]
        hash_to[h].append(f"{r}::{node.name}:{node.lineno}")

for h, members in hash_to.items():
    if len(members) < 2:
        continue
    # only record as B candidates when body length suggests real duplication not empty
    if len(members) >= 2:
        for m in members:
            add_b(
                m.split(":")[0] + ":" + m.split(":")[1] if False else m.rsplit(":", 1)[0],
                "exact_duplicate_body",
                "CANDIDATE",
                f"与同哈希用例重复 body hash={h}; peers={','.join(members)}",
                int(m.rsplit(":", 1)[1]),
                h,
            )

def write_tsv(path, rows, cols):
    with path.open("w", encoding="utf-8") as f:
        f.write("\t".join(cols) + "\n")
        for row in rows:
            f.write("\t".join(str(row.get(c, "")) for c in cols) + "\n")


write_tsv(OUT / f"{LABEL}-class-a-candidates.tsv", class_a, cols_a)
write_tsv(OUT / f"{LABEL}-class-b-candidates.tsv", class_b, cols_b)

summary["a_by_kind"] = dict(summary["a_by_kind"])
summary["b_flag_counts"] = dict(summary["b_flag_counts"])
(OUT / f"{LABEL}-enum-summary.json").write_text(

)
print(json.dumps(summary, ensure_ascii=False, indent=2))
