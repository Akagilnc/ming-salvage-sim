def py_files() -> list[Path]:
    out = []
    for rel in TRACKED:
        if not rel.endswith(".py"):

def collect_defs_and_refs() -> tuple[dict[str, list[str]], dict[str, list[str]]]:
    """name -> list of path:line for defs; name -> list of path:line for refs."""
    defs: dict[str, list[str]] = defaultdict(list)
    refs: dict[str, list[str]] = defaultdict(list)
    for path in py_files():
        rel = str(path.relative_to(ROOT))
        try:
            src = path.read_text(encoding="utf-8")
            tree = ast.parse(src, filename=rel)
        except Exception:
            continue
        for node in ast.walk(tree):
            if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
                defs[node.name].append(f"{rel}:{node.lineno}")
            if isinstance(node, ast.Name) and isinstance(node.ctx, ast.Load):
                refs[node.id].append(f"{rel}:{node.lineno}")
            if isinstance(node, ast.Attribute) and isinstance(node.ctx, ast.Load):
                refs[node.attr].append(f"{rel}:{node.lineno}")
    return defs, refs

                        "snippet": line.strip()[:200],
                    }
                )
    return rows


def hist_exclusive_candidates(seeds: list[str], defs, refs) -> list[dict]:
    """Downward exclusive-support candidates from historically deleted consumers.

    Exclusive := currently defined in repo AND all refs are under tests/ OR
    only mutual refs among seed/closure names (dead tree), OR asset/load binding
    with no live prod call outside the retired tree.
    """
    rows = []
    seed_set = set(seeds)
    # Expand one hop: names defined in same files as seed defs that are only test-ref'd
    closure = set(seed_set)
    for name in list(seed_set):
        for loc in defs.get(name, []):
            fpath = loc.split(":")[0]
            # siblings in file
            try:
                tree = ast.parse((ROOT / fpath).read_text(encoding="utf-8"))
            except Exception:
                continue
            for node in ast.walk(tree):
                if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
                    closure.add(node.name)

    for name in sorted(closure):
        dlocs = defs.get(name, [])
        rlocs = refs.get(name, [])
        if not dlocs and name not in seed_set:
            continue
        prod_refs = [r for r in rlocs if not r.startswith("tests/")]
        test_refs = [r for r in rlocs if r.startswith("tests/")]
        prod_defs = [d for d in dlocs if not d.startswith("tests/")]
        # Candidate if: hist seed OR (has prod def with zero non-test refs outside def sites)
        is_seed = name in seed_set
        exclusive_test = bool(prod_defs) and not any(
            r for r in prod_refs if r not in dlocs
        )
        if is_seed or exclusive_test or (not prod_defs and test_refs and name in seed_set):
            rows.append(
                {
                    "class": "F2-R13-1",
                    "kind": "hist_chase",
                    "path": (prod_defs or dlocs or test_refs or [""])[0].split(":")[0]
                    if (prod_defs or dlocs or test_refs)
                    else "",
                    "line": (prod_defs or dlocs or test_refs or ["?"])[0].split(":")[-1]
                    if (prod_defs or dlocs or test_refs)
                    else "",
                    "symbol": name,
                    "snippet": f"defs={len(dlocs)};prod_refs={len(prod_refs)};test_refs={len(test_refs)};seed={is_seed};exclusive_prod_def={exclusive_test}",
                }
            )
    return rows


def unparse_expr(node: ast.AST) -> str:
    try:
        return ast.unparse(node)
    except Exception:
        return type(node).__name__


def iter_test_functions(tree: ast.AST):
    for node in ast.walk(tree):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)) and node.name.startswith(
            "test_"
        ):
            yield node


def collect_asserts_in_fn(fn: ast.AST) -> list[tuple[int, str, str]]:
    """Return (lineno, kind, text) for asserts inside a test function."""
    out = []
    for node in ast.walk(fn):
        if isinstance(node, ast.Assert):
            text = unparse_expr(node.test)
            out.append((node.lineno, "assert", text[:300]))
        elif isinstance(node, ast.Call):
            f = node.func
            names = []
            if isinstance(f, ast.Name):
                names.append(f.id)
            elif isinstance(f, ast.Attribute):
                names.append(f.attr)
                if isinstance(f.value, ast.Name):
                    names.append(f.value.id)
            joined = ".".join(names)
            if any(
                x in joined
                for x in (
                    "assertEqual",
                    "assertTrue",
                    "assertFalse",
                    "assertIn",
                    "assertNotIn",
                    "assertRaises",
                    "assertIs",
                    "assertIsNone",
                    "assertIsNotNone",
                    "assertAlmostEqual",
                    "assertCountEqual",
                    "assertRegex",
                    "assertNotRegex",
                    "raises",
                )
            ):
                try:
                    text = unparse_expr(node)
                except Exception:
                    text = joined
                kind = "pytest.raises" if "raises" in joined else "unittest_assert"
                out.append((node.lineno, kind, text[:300]))
    return out


def classify_assert_flags(text: str, fn_name: str) -> str:
    flags = []
    if WORDING_RE.search(text):
        # Chinese or wording-ish — flag for review; not auto-delete
        if re.search(r"['\"][\u4e00-\u9fff].{0,40}['\"]", text) and re.search(
            r"==|in |not in |match", text
        ):
            flags.append("wording_cmp")
    if SHAPE_RE.search(text):
        flags.append("shape_lock")
    if HELPER_RE.search(fn_name):
        flags.append("helper_fn")
    if "getsource" in text or "inspect." in text:
        flags.append("source_lock")
    return ",".join(flags)


def enum_class_b() -> tuple[list[dict], list[dict]]:
    fn_rows = []
    assert_rows = []
    # Duplicate detection: identical assert text within same file
    by_file_text: dict[str, list[tuple[str, int, str]]] = defaultdict(list)

    for path in sorted((ROOT / "tests").rglob("test_*.py")):
        rel = str(path.relative_to(ROOT))
        if rel not in TRACKED and not any(t == rel for t in TRACKED):
            # still include if under tests and exists
            pass
        try:
            src = path.read_text(encoding="utf-8")
            tree = ast.parse(src, filename=rel)
        except Exception as e:
            fn_rows.append(
                {
                    "class": "F2-R13-2",
                    "kind": "test_fn",
                    "path": rel,
                    "line": "0",
                    "symbol": f"<parse_error:{e}>",
                    "snippet": "",
                    "flags": "parse_error",
                }
            )
            continue
        for fn in iter_test_functions(tree):
            fn_rows.append(
                {
                    "class": "F2-R13-2",
                    "kind": "test_fn",
                    "path": rel,
                    "line": str(fn.lineno),
                    "symbol": fn.name,
                    "snippet": "",
                    "flags": "helper_fn" if HELPER_RE.search(fn.name) else "",
                }
            )
            for lineno, kind, text in collect_asserts_in_fn(fn):
                flags = classify_assert_flags(text, fn.name)
                assert_rows.append(
                    {
                        "class": "F2-R13-2",
                        "kind": kind,
                        "path": rel,
                        "line": str(lineno),
                        "symbol": f"{fn.name}@{lineno}",
                        "snippet": text.replace("\t", " "),
                        "flags": flags,
                    }
                )
                by_file_text[rel].append((fn.name, lineno, text))

    # Mark duplicates within file (same assert text appearing >=2)
    for rel, items in by_file_text.items():
        counts = Counter(t for _, _, t in items)
        dup_texts = {t for t, c in counts.items() if c >= 2 and len(t) > 10}
        if not dup_texts:
            continue
        for row in assert_rows:
            if row["path"] == rel and row["snippet"] in dup_texts:
                flags = row["flags"].split(",") if row["flags"] else []
                if "duplicate" not in flags:
                    flags.append("duplicate")
                row["flags"] = ",".join(f for f in flags if f)

    return fn_rows, assert_rows


def write_tsv(path: Path, rows: list[dict], fields: list[str]) -> None:
    lines = ["\t".join(fields)]
    for r in rows:
        lines.append("\t".join(str(r.get(f, "")).replace("\t", " ").replace("\n", " ") for f in fields))
    path.write_text("\n".join(lines) + "\n", encoding="utf-8")


def main() -> None:
    seeds = load_hist_deleted_seeds()
    (OUT / "class-a-hist-seeds.txt").write_text("\n".join(seeds) + "\n", encoding="utf-8")

    defs, refs = collect_defs_and_refs()
    wide = wide_keyword_hits()
    hist = hist_exclusive_candidates(seeds, defs, refs)

    # AST prod defs whose name matches wide class-A surface
    ast_defs = []
    for name, locs in defs.items():
        if CLASS_A_WIDE.search(name) or name in set(seeds):
            for loc in locs:
                ast_defs.append(
                    {
                        "class": "F2-R13-1",
                        "kind": "ast_def",
                        "path": loc.split(":")[0],
                        "line": loc.split(":")[-1],
                        "symbol": name,
                        "snippet": f"refs={len(refs.get(name, []))}",
                    }
                )

    class_a = wide + hist + ast_defs
    # de-dup by path:line:symbol:kind
    seen = set()
    uniq = []
    for r in class_a:
        key = (r["kind"], r["path"], r["line"], r["symbol"], r["snippet"][:80])
        if key in seen:
            continue
        seen.add(key)
        uniq.append(r)
    class_a = uniq

    fn_rows, assert_rows = enum_class_b()

    fields_a = ["class", "kind", "path", "line", "symbol", "snippet"]
    fields_b = ["class", "kind", "path", "line", "symbol", "snippet", "flags"]
    write_tsv(OUT / "class-a-candidates.tsv", class_a, fields_a)
    write_tsv(OUT / "class-b-test-fns.tsv", fn_rows, fields_b)
    write_tsv(OUT / "class-b-asserts.tsv", assert_rows, fields_b)

    summary = {
        "class_a_candidates": len(class_a),
        "class_a_wide_kw": sum(1 for r in class_a if r["kind"] == "wide_kw"),
        "class_a_hist_chase": sum(1 for r in class_a if r["kind"] == "hist_chase"),
        "class_a_ast_def": sum(1 for r in class_a if r["kind"] == "ast_def"),
        "hist_seeds": len(seeds),
        "class_b_test_fns": len(fn_rows),
        "class_b_asserts": len(assert_rows),
        "class_b_flagged_wording": sum(1 for r in assert_rows if "wording_cmp" in r["flags"]),
        "class_b_flagged_shape": sum(1 for r in assert_rows if "shape_lock" in r["flags"]),
        "class_b_flagged_dup": sum(1 for r in assert_rows if "duplicate" in r["flags"]),
        "class_b_flagged_source": sum(1 for r in assert_rows if "source_lock" in r["flags"]),
    }
    (OUT / "enum-summary.json").write_text(
        json.dumps(summary, ensure_ascii=False, indent=2) + "\n", encoding="utf-8"
    )
    print(json.dumps(summary, ensure_ascii=False, indent=2))


if __name__ == "__main__":
    main()

def write_tsv(path: Path, rows, fields):
    with path.open("w", encoding="utf-8", newline="") as f:
        w = csv.DictWriter(f, fieldnames=fields, delimiter="\t", extrasaction="ignore", lineterminator="\n")
        w.writeheader()
        for r in rows:
            w.writerow({k: str(r.get(k, "")).replace("\t", " ").replace("\n", " ") for k in fields})
