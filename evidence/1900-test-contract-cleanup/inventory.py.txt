import ast
import collections
import hashlib
import json
import pathlib
import subprocess

BASE = 'f21ce4d05'
ROOT = pathlib.Path.cwd()
OUT = pathlib.Path('/tmp/1900-continuation-members')
OUT.mkdir(exist_ok=True)

baseline_paths = set(subprocess.check_output(['git', 'ls-tree', '-r', '--name-only', BASE], text=True).splitlines())
tracked = sorted(set(subprocess.check_output(['git', 'ls-files'], text=True).splitlines()) | baseline_paths)
paths = [p for p in tracked if p.endswith('.py') or (p.startswith('web/') and '.test.' in p and p.endswith(('.ts', '.tsx')))]
rows = []

node_program = r'''
const ts = require(process.cwd() + '/web/node_modules/typescript');
const fs = require('fs');
const input = JSON.parse(fs.readFileSync(0, 'utf8'));
const sf = ts.createSourceFile(input.path, input.source, ts.ScriptTarget.Latest, true,
  input.path.endsWith('.tsx') ? ts.ScriptKind.TSX : ts.ScriptKind.TS);
const out = [];
function rootName(expr) {
  if (ts.isIdentifier(expr)) return expr.text;
  if (ts.isPropertyAccessExpression(expr)) return rootName(expr.expression);
  if (ts.isCallExpression(expr)) return rootName(expr.expression);
  return '';
}
function visit(node, groups) {
  let next = groups;
  if (ts.isCallExpression(node)) {
    const root = rootName(node.expression);
    const callback = node.arguments.some(a => ts.isArrowFunction(a) || ts.isFunctionExpression(a));
    if (callback && ['it', 'test', 'describe'].includes(root)) {
      const label = node.arguments[0]?.getText(sf) || '<unnamed>';
      if (root === 'describe') next = groups.concat(label);
      else out.push({name: groups.concat(label).join(' / '), line: sf.getLineAndCharacterOfPosition(node.getStart(sf)).line + 1,
        endLine: sf.getLineAndCharacterOfPosition(node.end).line + 1, source: node.getText(sf)});
    }
  }
  ts.forEachChild(node, child => visit(child, next));
}
visit(sf, []);
process.stdout.write(JSON.stringify(out));
'''

def enumerate_members(path, source):
    if path.endswith('.py'):
        result = []
        def visit(node, scope):
            is_scope = isinstance(node, (ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef))
            name = scope + [node.name] if is_scope else scope
            if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)) and node.name.startswith('test_'):
                result.append({'name': '.'.join(name), 'line': node.lineno, 'endLine': node.end_lineno,
                    'source': ast.get_source_segment(source, node)})
            for child in ast.iter_child_nodes(node):
                visit(child, name)
        visit(ast.parse(source), [])
        return result
    return json.loads(subprocess.check_output(['node', '-e', node_program], input=json.dumps({'path': path, 'source': source}), text=True))

for path in paths:
    before_source = subprocess.check_output(['git', 'show', f'{BASE}:{path}'], text=True) if path in baseline_paths else ''
    versions = {'before': before_source,
                'after': (ROOT / path).read_text() if (ROOT / path).is_file() else ''}
    indexed = {}
    for version, source in versions.items():
        counts = collections.Counter()
        for item in enumerate_members(path, source):
            counts[item['name']] += 1
            key = (item['name'], counts[item['name']])
            item['sha256'] = hashlib.sha256(item.pop('source').encode()).hexdigest()
            indexed.setdefault(key, {'file': path, 'member': key[0], 'occurrence': key[1]})[version] = item
    for row in indexed.values():
        before, after = row.get('before'), row.get('after')
        row['disposition'] = 'removed' if after is None else 'added' if before is None else 'modified' if before['sha256'] != after['sha256'] else 'retained'
        rows.append(row)

rows.sort(key=lambda row: (row['file'], row['member'], row['occurrence']))
changed = [row for row in rows if row['disposition'] != 'retained']
summary = {'baseline': BASE, 'headBeforeCommit': subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip(),
    'enumeratedTrackedSourceFiles': len(paths), 'filesWithTestDeclarations': len({r['file'] for r in rows}),
    'beforeDeclarations': sum('before' in r for r in rows), 'afterDeclarations': sum('after' in r for r in rows),
    'dispositions': dict(collections.Counter(r['disposition'] for r in rows)),
    'changedFiles': len({r['file'] for r in changed}), 'changedDeclarations': len(changed)}
(OUT / 'all-members.json').write_text(json.dumps({'summary': summary, 'members': rows}, ensure_ascii=False, indent=2) + '\n')
(OUT / 'changed-members.json').write_text(json.dumps(changed, ensure_ascii=False, indent=2) + '\n')
with (OUT / 'changed-members.md').open('w') as output:
    output.write('# #1900 complete changed-member table\n\nBaseline: f21ce4d05; current working tree. Occurrence distinguishes shadowed duplicate definitions.\n\n| File | Member | Occurrence | Disposition | Before lines | After lines |\n|---|---|---:|---|---|---|\n')
    for row in changed:
        def span(version):
            item = row.get(version)
            return f"{item['line']}-{item['endLine']}" if item else '—'
        output.write(f"| `{row['file']}` | `{row['member'].replace('|', '&#124;')}` | {row['occurrence']} | {row['disposition']} | {span('before')} | {span('after')} |\n")
(OUT / 'summary.json').write_text(json.dumps(summary, ensure_ascii=False, indent=2) + '\n')
print(json.dumps(summary, ensure_ascii=False, indent=2))
