def walk_funcs(tree):
    out=[]
    def go(node, stack):
        for child in ast.iter_child_nodes(node):
            if isinstance(child, ast.ClassDef):
                go(child, stack+[child.name])
            elif isinstance(child, (ast.FunctionDef, ast.AsyncFunctionDef)):
                out.append(('.'.join(stack+[child.name]), child))
                go(child, stack+[child.name])
            else:
                go(child, stack)
    go(tree, [])
    return out

cands=[]

        if re.search(r'urgent:\{turn\}:', body) or re.search(r"f[\"']urgent:\{turn\}", body):
            shapes.append('urgent_fixture_id')
        if shapes:
            cands.append({'file':f,'member':q,'shapes':shapes,'lines':f'{node.lineno}-{node.end_lineno}'})
            print(f"{f}::{q} [{','.join(shapes)}] L{node.lineno}-{node.end_lineno}")
print('TOTAL', len(cands))
