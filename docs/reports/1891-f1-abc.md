import ast, json, re, subprocess
from pathlib import Path
ROOT = Path('/Users/akagilnc/WorkSpace/Ming_LLM-1899-w5')
tracked = subprocess.check_output(['git','ls-files','tests'], cwd=ROOT, text=True).splitlines()
files = [f for f in tracked if f.endswith('.py') and (ROOT/f).is_file()]
print(f'TRACKED_TEST_PY={len(files)}')

def walk_funcs(tree):
    out=[]
    def go(node, stack):
        for child in ast.iter_child_nodes(node):
            if isinstance(child, ast.ClassDef):
                go(child, stack+[child.name])
            elif isinstance(child, (ast.FunctionDef, ast.AsyncFunctionDef)):
                out.append(('.'.join(stack+[child.name]), child))
                go(child, stack+[child.name])
            else:
                go(child, stack)
    go(tree, [])
    return out

cands=[]
for f in files:
    src = (ROOT/f).read_text(encoding='utf-8', errors='replace')
    tree = ast.parse(src)
    lines = src.splitlines()

        body='\n'.join(lines[node.lineno-1:node.end_lineno])
        asserts=[]
        for sub in ast.walk(node):
            if isinstance(sub, ast.Assert):
                asserts.append(ast.get_source_segment(src, sub) or '')
        shapes=[]
        if 'assert not hasattr' in body or any('not hasattr' in (a or '') for a in asserts):
            shapes.append('hasattr_absence')
        if 'inspect.getsource' in body and 'not in' in body:
            shapes.append('source_absence')
        if 'ORDER BY CASE' in body:
            shapes.append('order_by_case_self')
        if any(('/api/' in (a or '') and 'not in' in (a or '')) for a in asserts) or (
            'registered_paths' in body and 'not in' in body and 'assert' in body):
            shapes.append('route_absence')
        if 'sqlite_master' in body and any(x in body for x in ['assert not','is None','== 0','not in']):
            shapes.append('sqlite_master_check')
        if 'PRAGMA table_info' in body:
            shapes.append('pragma_table_info')
        if re.search(r'ORDER BY .+ LIMIT', body, re.I) and 'conn.execute' in body and 'assert' in body:
            shapes.append('select_order_limit')
        if shapes:
            cands.append({'file':f,'member':q,'shapes':shapes,'lines':f'{node.lineno}-{node.end_lineno}'})
            print(f"{f}::{q} [{','.join(shapes)}] L{node.lineno}-{node.end_lineno}")
print('TOTAL', len(cands))
(ROOT/'docs/reports/1891-f1-abc-f1-candidates.json').write_text(
    json.dumps(cands, indent=2, ensure_ascii=False)+'\n', encoding='utf-8')
(ROOT/'docs/reports/1891-f1-abc-f1-candidates.txt').write_text(
    '\n'.join(f"{c['file']}::{c['member']} [{','.join(c['shapes'])}] L{c['lines']}" for c in cands)
