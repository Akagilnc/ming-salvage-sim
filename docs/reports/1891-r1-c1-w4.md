import ast, subprocess
from collections import defaultdict
from pathlib import Path

def ls_files():
    return [p for p in subprocess.check_output(['git','ls-files','*.py'], text=True).splitlines()
            if p.endswith('.py')]

defs = defaultdict(list)  # name -> [(path, kind, qual)]
refs = defaultdict(list)

class DefV(ast.NodeVisitor):
    def __init__(self, path):
        self.path = path; self.stack = []
    def visit_FunctionDef(self, n):
        defs[n.name].append((self.path, 'function', '.'.join(self.stack+[n.name])))
        self.stack.append(n.name); self.generic_visit(n); self.stack.pop()
    visit_AsyncFunctionDef = visit_FunctionDef
    def visit_ClassDef(self, n):
        defs[n.name].append((self.path, 'class', '.'.join(self.stack+[n.name])))
        self.stack.append(n.name); self.generic_visit(n); self.stack.pop()
    def _constish(self, name):
        return name.isupper() or (name.startswith('_') and any(c.isupper() for c in name[1:]))
    def visit_Assign(self, n):
        for t in n.targets:
            ids = ([t.id] if isinstance(t, ast.Name)
                   else ([e.id for e in t.elts if isinstance(e, ast.Name)]
                         if isinstance(t, ast.Tuple) else []))
            for name in ids:
                if self._constish(name):
                    defs[name].append((self.path, 'assign', name))
        self.generic_visit(n)
    def visit_AnnAssign(self, n):
        if isinstance(n.target, ast.Name) and self._constish(n.target.id):
            defs[n.target.id].append((self.path, 'annassign', n.target.id))
        self.generic_visit(n)

class RefV(ast.NodeVisitor):
    def __init__(self, path): self.path = path
    def visit_Name(self, n):
        if isinstance(n.ctx, ast.Load):
            refs[n.id].append((self.path, 'Load'))
        self.generic_visit(n)
    def visit_Attribute(self, n):
        refs[n.attr].append((self.path, 'Attribute')); self.generic_visit(n)
    def visit_ImportFrom(self, n):
        for a in n.names:
            refs[a.asname or a.name].append((self.path, 'import'))
            refs[a.name].append((self.path, 'import_name'))
        self.generic_visit(n)
    def visit_keyword(self, n):
        if n.arg: refs[n.arg].append((self.path, 'keyword'))
        self.generic_visit(n)

for rel in ls_files():
    src = Path(rel).read_text(encoding='utf-8')
    try:
        tree = ast.parse(src, filename=rel)
    except SyntaxError as e:
        raise SystemExit(f'PARSE FAIL {rel}: {e}') from e
    DefV(rel).visit(tree); RefV(rel).visit(tree)
